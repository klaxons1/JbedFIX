cmake_minimum_required(VERSION 3.16)
project(jbed_swerve LANGUAGES C CXX)

find_package(ZLIB REQUIRED)
option(JBED_FETCH_LIBPNG "Build the pinned libpng source instead of finding an installed library" ${ANDROID})
if(NOT TARGET PNG::PNG)
    if(JBED_FETCH_LIBPNG)
        include(FetchContent)
        if(POLICY CMP0135)
            cmake_policy(SET CMP0135 NEW)
        endif()
        function(jbed_build_libpng)
            set(PNG_SHARED OFF)
            set(PNG_STATIC ON)
            set(PNG_FRAMEWORK OFF)
            set(PNG_TESTS OFF)
            set(PNG_TOOLS OFF)
            FetchContent_Declare(jbed_libpng
                # libpng 1.6.59
                URL https://github.com/pnggroup/libpng/archive/cd952f49f95bb27154ae77dbb103032d95f6e580.tar.gz
                URL_HASH SHA256=ef1bccad2b2a58eb3c09e06ffe3b72b8604abda8b5011e4c4bb5391989040f26)
            FetchContent_MakeAvailable(jbed_libpng)
            set_target_properties(png_static PROPERTIES POSITION_INDEPENDENT_CODE ON)
            add_library(PNG::PNG ALIAS png_static)
        endfunction()
        jbed_build_libpng()
    else()
        find_package(PNG REQUIRED)
    endif()
endif()
set(kni_dir java/midp/src/share/native)
file(GLOB swerve_sources CONFIGURE_DEPENDS
    "*.c" "*.cpp"
    "rawapi/standard/*.cpp" "sys/*.cpp" "shared/*.c"
    "${kni_dir}/*.c" "${kni_dir}/standard/*.c")
add_library(jbedswerve STATIC ${swerve_sources})
target_compile_features(jbedswerve PUBLIC c_std_11 cxx_std_20)
target_include_directories(jbedswerve PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" shared
    PRIVATE "${kni_dir}" "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(jbedswerve PUBLIC ZLIB::ZLIB PNG::PNG)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(jbedswerve PUBLIC -fwrapv -ffp-contract=off)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(jbedswerve PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions>)
endif()
if(UNIX AND NOT APPLE)
    target_link_libraries(jbedswerve PUBLIC m)
endif()
